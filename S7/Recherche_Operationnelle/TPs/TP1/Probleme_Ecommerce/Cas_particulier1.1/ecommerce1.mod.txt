###############################  Model ###############################



###############################  Sets  ###############################

set PRODUITS; 

set MAGASINS;	

set COMMANDES;

################### Variables ###################

var Q{i in PRODUITS,j in MAGASINS , c in COMMANDES}, >=0;



###################  Constants: Data to load   #########################

param Coutsparproduit{j in MAGASINS,i in PRODUITS}; 

param Stock{j  in MAGASINS,i in PRODUITS };

param QuantiteDemandees {c in COMMANDES,i in PRODUITS};


################### Constraints ###################


s.t. RespectCapaciteDeStockMagasin{i in PRODUITS,j in MAGASINS}:
  sum{c in COMMANDES}  Q[i,j,c] <= Stock[j,i];
 
s.t. RespectQuantiteArriveesCelleDemandees{i in PRODUITS,c in COMMANDES}:
  sum{j in MAGASINS} Q[i,j,c] = QuantiteDemandees[c,i];


###### Objective ######

minimize Cout: 
		sum{i in PRODUITS, j in MAGASINS , c in COMMANDES} Q[i,j,c]*Coutsparproduit[j,i] ; 

#end;


#default data

data;

set PRODUITS := 
1
2;

set MAGASINS :=
M1
M2
M3;

set COMMANDES :=
C1
C2;

param Coutsparproduit: 1 2:=
M1 2 3
M2 2 1
M3 3 2;

param Stock: 1 2:=
M1 2.5 1
M2 3   2
M3 1   3;

param QuantiteDemandees: 1 2 :=
C1 1 1
C2 3 1;


end;
